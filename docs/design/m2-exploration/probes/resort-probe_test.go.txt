// Scratch probe, not committed: when does the party re-sort happen relative to
// order entry, and does an order follow its character or its slot?
func TestProbeResortTiming(t *testing.T) {
	names := []string{"ALDA", "BRAM", "CORA", "DREW", "EDDA", "FINN"}
	for s := int64(1); s < 40; s++ {
		seed(s)
		sc := loadScenario(t, 1)
		var ms []*Character
		for _, n := range names {
			ms = append(ms, &Character{Name: n, HP: 10, MaxHP: 10, Level: 1, Agility: 0})
		}
		ms[1].Status = Dead
		ms[1].HP = 0
		g := &GameState{Scenario: sc, Town: &TownState{Party: &Party{Members: ms}}}
		mon := &CombatMonster{MonsterID: 0, HP: 50, MaxHP: 50}
		cs := &CombatState{Phase: CombatConfirm, Round: 2, Actions: make([]PartyAction, 6),
			Groups: []*MonsterGroup{{MonsterID: 0, Members: []*CombatMonster{mon}, AliveCnt: 1}}}
		g.Combat = cs
		for i := range cs.Actions {
			cs.Actions[i] = PartyAction{Action: ActionParry}
		}
		cs.Actions[2] = PartyAction{Action: ActionRun} // entered for slot 3 = CORA
		before := []string{}
		for _, m := range ms {
			before = append(before, m.Name)
		}
		cs.ExecuteRound(g)
		after := []string{}
		for _, m := range g.Town.Party.Members {
			after = append(after, m.Name)
		}
		msg := strings.Join(cs.Messages, " | ")
		if strings.Contains(msg, "FAILED TO RUN") {
			t.Logf("seed %d\n order entry saw: %v\n after re-sort:   %v\n run order was given in slot 3\n messages: %s", s, before, after, msg)
			return
		}
	}
	t.Log("no failed-run seed found")
}

// Paralysis and sleep also move a member back, and the sort is not stable.
func TestProbeResortStatuses(t *testing.T) {
	mk := func(n string, st Status) *Character { return &Character{Name: n, Status: st} }
	cases := [][]*Character{
		{mk("A", OK), mk("B", Paralyzed), mk("C", OK), mk("D", OK), mk("E", OK), mk("F", OK)},
		{mk("A", OK), mk("B", Dead), mk("C", Paralyzed), mk("D", OK), mk("E", OK), mk("F", OK)},
		{mk("A", Asleep), mk("B", OK), mk("C", OK), mk("D", OK), mk("E", OK), mk("F", OK)},
		{mk("A", Dead), mk("B", OK), mk("C", OK), mk("D", Ashed), mk("E", OK), mk("F", Dead)},
	}
	for _, c := range cases {
		in := []string{}
		for _, m := range c {
			in = append(in, m.Name+":"+m.Status.String())
		}
		sortPartyByStatus(c)
		out := []string{}
		for _, m := range c {
			out = append(out, m.Name+":"+m.Status.String())
		}
		t.Logf("%v -> %v", in, out)
	}
}
